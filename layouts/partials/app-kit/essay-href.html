{{- /* Rewrite one href. Match rules follow static/js/essay-links.js: a dated blog.kindel.com permalink, or ?p= on the blog root, and only when the Essays category names that post. The result is https://kindel.com/essays// plus any fragment. Query strings are dropped. Every other href is unchanged. */ -}} {{- $href := printf "%s" . -}} {{- $out := $href -}} {{- $raw := $href -}} {{- if hasPrefix $raw "//" -}} {{- $raw = printf "https:%s" $raw -}} {{- end -}} {{- if or (hasPrefix $raw "https://") (hasPrefix $raw "http://") -}} {{- $parsed := try (urls.Parse $raw) -}} {{- if and (not $parsed.Err) $parsed.Value -}} {{- $u := $parsed.Value -}} {{- $host := lower $u.Host -}} {{- $host = $host | strings.TrimPrefix "www." -}} {{- if and (or (eq $u.Scheme "https") (eq $u.Scheme "http")) (eq $host "blog.kindel.com") -}} {{- $cat := partialCached "app-kit/essay-catalog.html" "catalog" -}} {{- $slug := "" -}} {{- $path := $u.Path | strings.TrimSuffix "/" | strings.TrimPrefix "/" -}} {{- $parts := split $path "/" -}} {{- if eq (len $parts) 4 -}} {{- $y := index $parts 0 -}} {{- $mo := index $parts 1 -}} {{- $day := index $parts 2 -}} {{- if and (findRE `^[0-9]{4}$` $y) (findRE `^[0-9]{2}$` $mo) (findRE `^[0-9]{2}$` $day) -}} {{- $candidate := partial "app-kit/essay-slug.html" (index $parts 3) -}} {{- if $candidate -}} {{- with index $cat.bySlug $candidate -}} {{- $slug = . -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- if not $slug -}} {{- $bare := or (eq $u.Path "/") (eq $u.Path "") (eq (lower $u.Path) "/index.php") -}} {{- if $bare -}} {{- $id := "" -}} {{- range split $u.RawQuery "&" -}} {{- if hasPrefix . "p=" -}} {{- $id = strings.TrimPrefix "p=" . -}} {{- end -}} {{- end -}} {{- if findRE `^[0-9]+$` $id -}} {{- $n := int $id -}} {{- if gt $n 0 -}} {{- with index $cat.byId (printf "%d" $n) -}} {{- $slug = . -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- if $slug -}} {{- $safe := partial "app-kit/essay-slug.html" $slug -}} {{- if $safe -}} {{- $out = printf "https://kindel.com/essays/%s/" $safe -}} {{- with $u.EscapedFragment -}} {{- $out = printf "%s#%s" $out . -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- return $out -}}
